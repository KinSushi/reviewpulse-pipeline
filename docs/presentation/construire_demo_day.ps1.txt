# --- Charte ---------------------------------------------------------------------------------------
function Couleur([string]$hex) { [int]('0x' + $hex.Substring(4, 2)) * 65536 + [int]('0x' + $hex.Substring(2, 2)) * 256 + [int]('0x' + $hex.Substring(0, 2)) }
$ENCRE  = Couleur '0E3449'   # titres
$GRIS   = Couleur '55575A'   # corps -- pas $TEXTE : PowerShell ignore la casse, et $texte est un parametre
$ACCENT = Couleur 'FF7047'   # bordure des cartes, mots-clés
$MENTHE = Couleur 'C3FFFC'   # bandeau bas du gabarit, fond des cartes de chiffres
$BLANC  = Couleur 'FFFFFF'
$POLICE_TITRE = 'Inter SemiBold'
$POLICE_TEXTE = 'Manrope'
$msoTrue = -1; $msoFalse = 0
$ppLayoutBlank = 12

function Texte($slide, [double]$x, [double]$y, [double]$w, [double]$h, [string]$texte, [double]$taille, [bool]$gras = $false, [int]$couleur = $GRIS, [string]$police = $POLICE_TEXTE, [int]$align = 1) {
    $f = $slide.Shapes.AddTextbox(1, $x, $y, $w, $h)
    $f.TextFrame.WordWrap = $msoTrue
    $f.TextFrame.AutoSize = 0
    $f.TextFrame.MarginLeft = 4; $f.TextFrame.MarginRight = 4; $f.TextFrame.MarginTop = 2; $f.TextFrame.MarginBottom = 2
    $r = $f.TextFrame.TextRange
    $r.Text = $texte
    $r.Font.Name = $police; $r.Font.Size = $taille; $r.Font.Bold = $(if ($gras) { $msoTrue } else { $msoFalse }); $r.Font.Color.RGB = $couleur
    $r.ParagraphFormat.Alignment = $align
    $r.ParagraphFormat.SpaceWithin = 1.05
    return $f
}
function Titre($slide, [string]$texte) {
    Texte $slide 90 22 594 40 $texte 24 $true $ENCRE $POLICE_TITRE | Out-Null
}
function Carte($slide, [double]$x, [double]$y, [double]$w, [double]$h, [string]$entete, [string[]]$lignes, [double]$taille = 11) {
    $c = $slide.Shapes.AddShape(5, $x, $y, $w, $h)   # rectangle arrondi
    $c.Fill.ForeColor.RGB = $BLANC; $c.Line.ForeColor.RGB = $ACCENT; $c.Line.Weight = 0.75
    $c.Adjustments.Item(1) = 0.06
    $c.TextFrame.MarginLeft = 8; $c.TextFrame.MarginRight = 8; $c.TextFrame.MarginTop = 5; $c.TextFrame.MarginBottom = 5
    $c.TextFrame.WordWrap = $msoTrue; $c.TextFrame.AutoSize = 0
    $c.TextFrame.VerticalAnchor = 1
    $r = $c.TextFrame.TextRange
    $r.Text = (@($entete) + $lignes) -join "`r"
    $r.Font.Name = $POLICE_TEXTE; $r.Font.Size = $taille; $r.Font.Color.RGB = $GRIS
    $r.ParagraphFormat.Alignment = 1; $r.ParagraphFormat.SpaceWithin = 1.05; $r.ParagraphFormat.SpaceBefore = 2
    $t = $r.Paragraphs(1); $t.Font.Bold = $msoTrue; $t.Font.Color.RGB = $ENCRE; $t.Font.Size = $taille + 1
    return $c
}
function Puces($slide, [double]$x, [double]$y, [double]$w, [double]$h, [string[]]$lignes, [double]$taille = 11.5) {
    $f = Texte $slide $x $y $w $h ($lignes -join "`r") $taille $false $GRIS
    $r = $f.TextFrame.TextRange
    $r.ParagraphFormat.Bullet.Visible = $msoTrue
    $r.ParagraphFormat.Bullet.Character = 8226
    $r.ParagraphFormat.Bullet.Font.Color.RGB = $ACCENT
    $r.ParagraphFormat.SpaceBefore = 4
    $r.IndentLevel = 1
    $f.TextFrame.Ruler.Levels(1).FirstMargin = 0; $f.TextFrame.Ruler.Levels(1).LeftMargin = 14
    return $f
}
function SousTitre($slide, [string]$texte, [double]$y = 66) {
    Texte $slide 36 $y 648 22 $texte 13 $true $GRIS | Out-Null
}
function Chiffre($slide, [double]$x, [double]$y, [double]$w, [string]$valeur, [string]$libelle) {
    $c = $slide.Shapes.AddShape(5, $x, $y, $w, 62)
    $c.Fill.ForeColor.RGB = $MENTHE; $c.Line.Visible = $msoFalse; $c.Adjustments.Item(1) = 0.1
    $c.TextFrame.MarginTop = 3; $c.TextFrame.MarginBottom = 3; $c.TextFrame.WordWrap = $msoTrue; $c.TextFrame.AutoSize = 0
    $r = $c.TextFrame.TextRange
    $r.Text = $valeur + "`r" + $libelle
    $r.ParagraphFormat.Alignment = 2
    $r.Font.Name = $POLICE_TEXTE; $r.Font.Color.RGB = $ENCRE
    $r.Paragraphs(1).Font.Size = 20; $r.Paragraphs(1).Font.Bold = $msoTrue; $r.Paragraphs(1).Font.Name = $POLICE_TITRE
    $r.Paragraphs(2).Font.Size = 9.5; $r.Paragraphs(2).Font.Color.RGB = $GRIS
}
function Bandeau($slide) {
    # Le liseré menthe du bas de page, comme dans le gabarit.
    $b = $slide.Shapes.AddShape(1, 0, 385, 720, 20); $b.Fill.ForeColor.RGB = $MENTHE; $b.Line.Visible = $msoFalse
}
function Logo($slide, $modele) {
    # Le petit logo Jedha en haut à gauche : copié depuis une diapositive de contenu du gabarit.
    $modele.Copy(); $p = $slide.Shapes.Paste(); $p.Left = 36; $p.Top = 26; $p.Width = 45; $p.Height = 30
}

    $nouvelles = @()

    if (-not $dejaOuvert) { $pp.Quit() }
